## Go-Starter Deployment Patterns

### Blueprint-Specific Playbooks

#### CLI Application Deployment
```yaml
---
# cli-deployment.yml
- name: Deploy CLI Application
  hosts: deployment_targets
  become: yes
  vars:
    cli_name: "{{ project_name }}"
    cli_version: "{{ app_version | default('latest') }}"
    install_path: "/usr/local/bin"
    
  tasks:
    - name: Create application user
      user:
        name: "{{ cli_name }}"
        system: yes
        shell: /bin/false
        home: "/opt/{{ cli_name }}"
        create_home: yes
      
    - name: Download CLI binary
      get_url:
        url: "{{ binary_download_url }}"
        dest: "{{ install_path }}/{{ cli_name }}"
        mode: '0755'
        owner: root
        group: root
      notify: restart_service
      
    - name: Create systemd service file
      template:
        src: cli-service.j2
        dest: "/etc/systemd/system/{{ cli_name }}.service"
        mode: '0644'
      notify: 
        - reload_systemd
        - restart_service
        
    - name: Enable and start service
      systemd:
        name: "{{ cli_name }}"
        enabled: yes
        state: started
        daemon_reload: yes
        
  handlers:
    - name: reload_systemd
      systemd:
        daemon_reload: yes
        
    - name: restart_service
      systemd:
        name: "{{ cli_name }}"
        state: restarted
```

#### Web API Deployment
```yaml
---
# web-api-deployment.yml
- name: Deploy Web API Application
  hosts: web_servers
  become: yes
  vars:
    app_name: "{{ project_name }}"
    app_port: "{{ api_port | default(8080) }}"
    app_environment: "{{ environment | default('production') }}"
    
  roles:
    - role: nginx
      vars:
        nginx_sites:
          - server_name: "{{ api_domain }}"
            listen: 80
            listen_ssl: 443
            location_blocks:
              - location: "/"
                proxy_pass: "http://localhost:{{ app_port }}"
                proxy_set_headers:
                  - "Host $host"
                  - "X-Real-IP $remote_addr"
                  - "X-Forwarded-For $proxy_add_x_forwarded_for"
                  - "X-Forwarded-Proto $scheme"
                  
    - role: database
      vars:
        db_name: "{{ app_name }}_{{ app_environment }}"
        db_user: "{{ app_name }}_user"
        db_password: "{{ vault_db_password }}"
        
    - role: application
      vars:
        app_config_template: "web-api-config.j2"
        app_binary_url: "{{ binary_download_url }}"
        environment_variables:
          DB_HOST: "{{ db_host }}"
          DB_PORT: "{{ db_port }}"
          DB_NAME: "{{ db_name }}"
          DB_USER: "{{ db_user }}"
          DB_PASSWORD: "{{ vault_db_password }}"
          API_PORT: "{{ app_port }}"
          LOG_LEVEL: "{{ log_level | default('info') }}"
```

#### Lambda Function Deployment
```yaml
---
# lambda-deployment.yml
- name: Deploy Lambda Function
  hosts: localhost
  connection: local
  vars:
    function_name: "{{ project_name }}"
    lambda_runtime: "go1.x"
    lambda_handler: "main"
    
  tasks:
    - name: Build Go Lambda binary
      shell: |
        cd {{ source_directory }}
        GOOS=linux GOARCH=amd64 go build -o main main.go
      delegate_to: localhost
      
    - name: Create deployment package
      archive:
        path: "{{ source_directory }}/main"
        dest: "/tmp/{{ function_name }}-{{ ansible_date_time.epoch }}.zip"
        format: zip
      register: lambda_package
      
    - name: Deploy Lambda function
      amazon.aws.lambda:
        name: "{{ function_name }}"
        zip_file: "{{ lambda_package.dest }}"
        runtime: "{{ lambda_runtime }}"
        handler: "{{ lambda_handler }}"
        memory_size: "{{ lambda_memory | default(128) }}"
        timeout: "{{ lambda_timeout | default(30) }}"
        environment_variables: "{{ lambda_env_vars | default({}) }}"
        tags: "{{ lambda_tags | default({}) }}"
      register: lambda_result
      
    - name: Configure Lambda triggers
      amazon.aws.lambda_event:
        function_name: "{{ function_name }}"
        event_source: "{{ item.source }}"
        source_params: "{{ item.params }}"
      loop: "{{ lambda_triggers | default([]) }}"
```

#### gRPC Gateway Deployment
```yaml
---
# grpc-gateway-deployment.yml
- name: Deploy gRPC Gateway Service
  hosts: grpc_servers
  become: yes
  vars:
    gateway_name: "{{ project_name }}"
    grpc_port: "{{ grpc_port | default(9090) }}"
    http_port: "{{ http_port | default(8080) }}"
    
  tasks:
    - name: Deploy gRPC service binary
      copy:
        src: "{{ binary_path }}"
        dest: "/opt/{{ gateway_name }}/bin/{{ gateway_name }}"
        mode: '0755'
        owner: "{{ gateway_name }}"
        group: "{{ gateway_name }}"
      notify: restart_grpc_service
      
    - name: Configure TLS certificates
      copy:
        content: "{{ item.content }}"
        dest: "/opt/{{ gateway_name }}/certs/{{ item.name }}"
        mode: '0600'
        owner: "{{ gateway_name }}"
        group: "{{ gateway_name }}"
      loop:
        - name: server.crt
          content: "{{ vault_tls_certificate }}"
        - name: server.key
          content: "{{ vault_tls_private_key }}"
      notify: restart_grpc_service
      
    - name: Configure load balancer for dual protocols
      template:
        src: grpc-gateway-nginx.j2
        dest: /etc/nginx/sites-available/{{ gateway_name }}
        backup: yes
      notify: reload_nginx
      
    - name: Enable site
      file:
        src: /etc/nginx/sites-available/{{ gateway_name }}
        dest: /etc/nginx/sites-enabled/{{ gateway_name }}
        state: link
      notify: reload_nginx
```

### Reusable Application Role
```yaml
# roles/go-application/tasks/main.yml
---
- name: Include OS-specific variables
  include_vars: "{{ ansible_os_family }}.yml"

- name: Create application user and group
  group:
    name: "{{ app_group }}"
    system: yes
  
- name: Create application user
  user:
    name: "{{ app_user }}"
    group: "{{ app_group }}"
    system: yes
    shell: /bin/false
    home: "{{ app_home }}"
    create_home: yes

- name: Create application directories
  file:
    path: "{{ item }}"
    state: directory
    owner: "{{ app_user }}"
    group: "{{ app_group }}"
    mode: '0755'
  loop:
    - "{{ app_home }}/bin"
    - "{{ app_home }}/config"
    - "{{ app_home }}/logs"
    - "{{ app_home }}/data"

- name: Install application binary
  get_url:
    url: "{{ app_binary_url }}"
    dest: "{{ app_home }}/bin/{{ app_name }}"
    mode: '0755'
    owner: "{{ app_user }}"
    group: "{{ app_group }}"
  notify: restart application

- name: Generate application configuration
  template:
    src: "{{ app_config_template }}"
    dest: "{{ app_home }}/config/{{ app_name }}.yaml"
    owner: "{{ app_user }}"
    group: "{{ app_group }}"
    mode: '0600'
  notify: restart application

- name: Create systemd service file
  template:
    src: systemd-service.j2
    dest: "/etc/systemd/system/{{ app_name }}.service"
    mode: '0644'
  notify: 
    - reload systemd
    - restart application

- name: Enable and start application service
  systemd:
    name: "{{ app_name }}"
    enabled: yes
    state: started
    daemon_reload: yes
```

## Security and Compliance

### Security Hardening Playbook
```yaml
---
# security-hardening.yml
- name: Security Hardening
  hosts: all
  become: yes
  
  roles:
    - role: security
      vars:
        security_ssh_port: 2222
        security_ssh_permit_root_login: "no"
        security_ssh_password_authentication: "no"
        security_fail2ban_enabled: true
        security_ufw_enabled: true
        security_automatic_updates: true
        
  tasks:
    - name: Configure firewall rules
      ufw:
        rule: "{{ item.rule }}"
        port: "{{ item.port }}"
        proto: "{{ item.proto | default('tcp') }}"
        from_ip: "{{ item.from_ip | default('any') }}"
      loop: "{{ firewall_rules }}"
      
    - name: Install security packages
      package:
        name: "{{ item }}"
        state: present
      loop:
        - fail2ban
        - ufw
        - aide
        - rkhunter
        - chkrootkit
        
    - name: Configure log rotation
      template:
        src: logrotate.j2
        dest: /etc/logrotate.d/{{ app_name }}
        mode: '0644'
```

## Deployment Strategies

### Blue-Green Deployment
```yaml
---
# blue-green-deployment.yml
- name: Blue-Green Deployment
  hosts: "{{ target_environment }}_servers"
  serial: "{{ deployment_batch_size | default(1) }}"
  
  vars:
    deployment_color: "{{ 'blue' if current_color == 'green' else 'green' }}"
    
  tasks:
    - name: Deploy to {{ deployment_color }} environment
      include_role:
        name: go-application
      vars:
        app_port: "{{ blue_port if deployment_color == 'blue' else green_port }}"
        app_config_suffix: "-{{ deployment_color }}"
        
    - name: Health check {{ deployment_color }} deployment
      uri:
        url: "http://{{ inventory_hostname }}:{{ app_port }}/health"
        method: GET
        status_code: 200
      retries: 10
      delay: 5
      
    - name: Switch load balancer to {{ deployment_color }}
      template:
        src: nginx-upstream.j2
        dest: /etc/nginx/conf.d/upstream.conf
        backup: yes
      vars:
        active_color: "{{ deployment_color }}"
      notify: reload nginx
      run_once: true
      delegate_to: "{{ groups['load_balancers'][0] }}"
      
    - name: Stop {{ 'green' if deployment_color == 'blue' else 'blue' }} services
      systemd:
        name: "{{ app_name }}-{{ 'green' if deployment_color == 'blue' else 'blue' }}"
        state: stopped
      when: switch_successful is defined and switch_successful
```

### Rolling Updates
```yaml
---
# rolling-update.yml
- name: Rolling Update Deployment
  hosts: web_servers
  serial: "{{ rolling_update_batch_size | default('25%') }}"
  max_fail_percentage: 10
  
  pre_tasks:
    - name: Remove server from load balancer
      uri:
        url: "{{ load_balancer_api }}/servers/{{ inventory_hostname }}/disable"
        method: POST
      delegate_to: localhost
      
  roles:
    - role: go-application
      vars:
        app_version: "{{ new_app_version }}"
        
  post_tasks:
    - name: Health check after update
      uri:
        url: "http://{{ inventory_hostname }}:{{ app_port }}/health"
        method: GET
        status_code: 200
      retries: 5
      delay: 10
      
    - name: Add server back to load balancer
      uri:
        url: "{{ load_balancer_api }}/servers/{{ inventory_hostname }}/enable"
        method: POST
      delegate_to: localhost
```
