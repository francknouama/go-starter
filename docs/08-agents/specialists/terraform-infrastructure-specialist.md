## Go-Starter Infrastructure Patterns

### Blueprint-Specific Infrastructure

#### CLI Applications
```hcl
# CLI binary distribution infrastructure
module "cli_distribution" {
  source = "./modules/cli-distribution"
  
  project_name = var.project_name
  binary_name  = var.binary_name
  platforms    = ["linux", "darwin", "windows"]
  
  # S3 bucket for binary distribution
  distribution_bucket = var.distribution_bucket
  
  # CloudFront for global distribution
  enable_cdn = var.enable_cdn
  
  tags = var.common_tags

```

#### Web API Infrastructure
```hcl
# Scalable web API infrastructure
module "web_api" {
  source = "./modules/web-api"
  
  project_name = var.project_name
  environment  = var.environment
  
  # Application Load Balancer
  alb_config = {
    certificate_arn = var.ssl_certificate_arn
    health_check_path = "/health"
  }
  
  # ECS Service
  ecs_config = {
    cpu    = var.api_cpu
    memory = var.api_memory
    desired_count = var.api_replicas
  }
  
  # RDS Database
  database_config = {
    engine         = var.db_engine
    instance_class = var.db_instance_class
    multi_az      = var.environment == "production"
  }
  
  # Redis Cache
  redis_config = {
    node_type = var.redis_node_type
    num_cache_nodes = var.redis_replicas
  }
  
  tags = var.common_tags
}
```

#### Lambda Function Infrastructure
```hcl
# Serverless Lambda infrastructure
module "lambda_function" {
  source = "./modules/lambda"

}
```

### Workspace Configuration
```hcl
# terraform.tf
terraform {
  required_version = ">= 1.0"
  
  required_providers {
    aws = {
      source  = "hashicorp/aws"
      version = "~> 5.0"
    }
  }
  
  backend "s3" {
    bucket         = "go-starter-terraform-state"
    key            = "${var.project_name}/${var.environment}/terraform.tfstate"
    region         = "us-west-2"
    encrypt        = true
    dynamodb_table = "terraform-state-locks"
  }
}
```

## Security and Compliance

### Security Best Practices
- **Least Privilege IAM**: Minimal required permissions for each component
- **Encryption at Rest**: All storage encrypted with KMS keys
- **Encryption in Transit**: TLS/SSL for all communications
- **Network Security**: Private subnets, security groups, NACLs
- **Secrets Management**: No hardcoded secrets, proper secret rotation
- **Audit Logging**: CloudTrail, VPC Flow Logs, application logging

## Cost Optimization

### Resource Right-Sizing
```hcl
# Cost optimization locals
locals {
  # Environment-based sizing
  instance_sizes = {
    development = "t3.micro"
    staging     = "t3.small"
    production  = "t3.medium"
  }
  
  # Auto-scaling based on environment
  scaling_config = {
    development = { min = 1, max = 2, desired = 1 }
    staging     = { min = 1, max = 4, desired = 2 }
    production  = { min = 2, max = 10, desired = 3 }
  }
}
```

### Spot Instances and Reserved Instances
```hcl
# Mixed instance policy for cost optimization
resource "aws_autoscaling_group" "app" {
  name = "${var.project_name}-${var.environment}-asg"
  
  mixed_instances_policy {
    launch_template {
      launch_template_specification {
        launch_template_id = aws_launch_template.app.id
        version           = "$Latest"
      }
      
      override {
        instance_type     = "t3.medium"
        weighted_capacity = "1"
      }
      
      override {
        instance_type     = "t3.large"
        weighted_capacity = "2"
      }
    }
    
    instances_distribution {
      on_demand_base_capacity                  = 1
      on_demand_percentage_above_base_capacity = 25
      spot_allocation_strategy                 = "diversified"
    }
  }
}
```
